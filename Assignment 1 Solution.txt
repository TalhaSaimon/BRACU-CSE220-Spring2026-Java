//Task-1(Row Rotation Policy)


    public static Integer rowRotation( Integer examWeek, String[][] matrix ){

      for(int i =0;i<examWeek-1;i++){
        for(int col =0;col<matrix[0].length;col++){
          String temp = matrix[matrix.length-1][col];
          for(int row=matrix.length-1;row>=1;row--){
            matrix[row][col]=matrix[row-1][col];
          }
          matrix[0][col]=temp;
        }
      }
      int row=1;
      for(int i =0;i<matrix.length;i++){
        for(int j=0;j<matrix[0].length;j++){
          System.out.print("| "+matrix[i][j]+"  ");
          if(matrix[i][j].equals("AA")){
            row+=i;
          }
        }
        System.out.println();
      }
       

        return row;
    }

//Task-3(Game Arena)



    public static void playGame( Integer[][] arena ){
      int sum =0;
      for(int i=0;i<arena.length;i++){
        for(int j=0;j<arena[0].length;j++){
          if(arena[i][j]%50==0 && arena[i][j]!=0){
            if(i<arena.length-1){
              if(arena[i+1][j]==2){
                sum +=2;
              }
            }
            if(i>0){
              if(arena[i-1][j]==2){
                sum +=2;
              }
            }
            if(j<arena[0].length-1){
               if(arena[i][j+1]==2){
                sum +=2;
              }
            }
            if(j>0){
              if(arena[i][j-1]==2){
                sum +=2;
              }
            }
          }
        }
      }
      if(sum>=10){
         System.out.println("Points Gained: "+sum+". Your team has survived the game.");
      }
      else{
        System.out.println("Points Gained: "+sum+". Your team is out.");
      }

    }



//Task-4(Rotate Secret)



    public static void rotateSecret( Character[][] board ){
      int len=board.length;
      int n=len/2;
      int left=0,right=len-1,top=0,bottom=len-1;
      while(n>=1){
       
        for(int i=0;i<n;i++){
           char temp=board[top][left];
           for(int a =top;a<bottom;a++){
             board [a][left]=board [a+1][left];
           }
           for(int b =left;b<right;b++){
              board [bottom][b]=board [bottom][b+1];
           }
           for(int c =bottom;c>top;c--){
              board [c][right]=board [c-1][right];
           }
           for(int d =right;d>left+1;d--){
              board [top][d]=board [top][d-1];
           }
           board [top][left+1]=temp;
        }
        n--;
        top++;
        bottom--;
        left++;
        right--;
      }
      String s="";
      for(int i =0;i<board.length;i++){
        for(int j=0;j<board[0].length;j++){
          s += board[i][j];
        }
      }
      System.out.println(s);
      

      }
