
    public static String checkSimilar( Node building1, Node building2 ){
      while(building1!=null && building2 !=null){
        if(!building1.elem.equals(building2.elem)){
          return "Not Similar";
        }
        else{
          building1=building1.next;
          building2=building2.next;
        }
      }
      if(building1!=null || building2 !=null){
        return "Not Similar";
      }
      else{
      return "Similar";
      }
    }

      Node curr=head;

      while(true){
        max=getmax(curr,popularity,track);
        int temp=popularity[max];
        popularity[max]=popularity[track];
        popularity[track]=temp;
        int i=track;
        Node curr1=curr;
        while(curr1!=null){
        if(i==max){
          Object e=curr.elem;
          curr.elem=curr1.elem;
          curr1.elem=e;

    public static int getmax(Node curr,Integer[]arr,int idx){
       Node curr1=curr;
       int max=Integer.MIN_VALUE;
       int i=idx;
       int maxIdx = 0;
      while(curr1!=null){
        if(arr[i]>max){
          max=arr[i];
          maxIdx=i;  
        }
        curr1=curr1.next;
        i++;
      }
      return maxIdx;
    }

    public static void sumOddAppend(Node dh) {
        Node even=dh;
        Node odd=dh.next;
        int sum=0;
        while(even.next!=dh && odd.next!=dh){
          sum +=(int)odd.elem;
          odd=odd.next.next;
          even.next=even.next.next;
          even=even.next;
        }
        even.next=new Node(sum);
        even.next.next=dh;
        
    }



public static void pairJoin(DNode dh1, DNode dh2) {
      DNode curr1=dh1.next;
      DNode curr2=dh2.next;
      while(curr2.next != null){

        DNode temp1=curr1.next;
        curr1.next=curr2;
        curr2.prev=curr1;
        DNode temp2=curr2.next;
        curr2.next=temp1;
        temp1.prev=curr2;
        curr1=temp1;
        curr2=temp2;
      }
      curr1.next=curr2;
      curr2.next=curr1;
      curr2.next = dh1;
      dh1.prev = curr2;
    }


 public static void rangeMove(DNode dh, int start, int end) {
        DNode curr=dh.next;
        int n=0;
        while(curr!=dh){
          curr=curr.next;
          n++;
        }
        curr=dh.next;
        
        int i=0;
        while(i<n){
          if((int)curr.elem>=start && (int)curr.elem<=end){
            DNode temp=curr.next;
            curr.next.prev=curr.prev;
            curr.prev.next=curr.next;
            
            curr.prev=dh.prev;
            dh.prev.next=curr;
            curr.next=dh;
            dh.prev=curr;
            curr=temp;
          }
          else{
            curr=curr.next;
          }
          i++;
        }
